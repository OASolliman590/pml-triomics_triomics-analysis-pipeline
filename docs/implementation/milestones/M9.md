```sh
(
  set -eu
  test "$(cat "$SOURCE_ARCHIVE/inst/triomics-source-revision")" = "$FINAL_COMMIT"
  test ! -e "$SOURCE_COPY"
  test ! -e "$VERIFY_ROOT"
  test ! -e "$VERIFY_LIB"
  mkdir -p "$SOURCE_COPY" "$VERIFY_ROOT/html" "$VERIFY_ROOT/walkthrough-runs" "$VERIFY_ROOT/native" "$VERIFY_LIB"
  export R_LIBS="$VERIFY_LIB${R_LIBS:+:$R_LIBS}"
  export R_LIBS_USER="$VERIFY_LIB${R_LIBS_USER:+:$R_LIBS_USER}"
  R CMD INSTALL --library="$VERIFY_LIB" "$SOURCE_ARCHIVE"
  cmp "$SOURCE_ARCHIVE/inst/triomics-source-revision" "$VERIFY_LIB/triomics/triomics-source-revision"
  Rscript --vanilla -e 'suppressPackageStartupMessages(library(triomics)); stopifnot(identical(normalizePath(find.package("triomics")), normalizePath(commandArgs(TRUE)[1])))' "$VERIFY_LIB/triomics"
  cp -R "$SOURCE_ARCHIVE/." "$SOURCE_COPY/"
  chmod -R u+w "$SOURCE_COPY"
  diff -qr "$SOURCE_ARCHIVE" "$SOURCE_COPY"
  export TRIOMICS_PACKET_EVIDENCE_ROOT="$VERIFY_ROOT/walkthrough-runs"
  export TRIOMICS_D_T09_PROJECT_ROOT="$SOURCE_COPY"
  cd "$SOURCE_COPY"
  for design in paired independent; do
    "$QUARTO_PATH" render "vignettes/$design-study.qmd" --to html --embed-resources --output-dir "$VERIFY_ROOT/html"
  done
  cd "$SOURCE_ARCHIVE"
  for stage in A B C D report; do
    Rscript --vanilla inst/exec/triomics --config inst/examples/tickets/D-T09.yaml --stage "$stage" --output "$VERIFY_ROOT/native"
  done
)
```
